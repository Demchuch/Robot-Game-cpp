EnemyController::EnemyController( std::vector<Robot> &robots, MovementSystem &movement)
	: robots_(robots), movement_(movement){}



void EnemyController::AiTurn() {
	for (Robot &robot : robots_) {
		if (robot.GetTeam() != Team::kPlayer && robot.GetHealth() > 0) {
			Robot *target = FindNearestHostile(robot);
			if (target == nullptr) {
				continue;
			}
			int row_diff = target->GetRow() - robot.GetRow();
			int col_diff = target->GetCol() - robot.GetCol();

			bool row_is_primary = std::abs(row_diff) > std::abs(col_diff);

			if (row_is_primary) {
				int next_row = robot.GetRow() + (row_diff > 0 ? 1 : -1);
				int next_col = robot.GetCol();
				bool moved = movement_.Action(robot, FindRobotAt(next_row, next_col), next_row, next_col);
				if (!moved && col_diff != 0) {
					int side_col = robot.GetCol() + (col_diff > 0 ? 1 : -1);
					movement_.Action(robot, FindRobotAt(robot.GetRow(), side_col), robot.GetRow(),
									side_col);
				}
			} else {
				int next_row = robot.GetRow();
				int next_col = robot.GetCol() + (col_diff > 0 ? 1 : -1);
				bool moved = movement_.Action(robot, FindRobotAt(next_row, next_col),  next_row, next_col);
				if (!moved && row_diff != 0) {
					int side_row = robot.GetRow() + (row_diff > 0 ? 1 : -1);
					movement_.Action(robot, FindRobotAt(side_row, robot.GetCol()),  side_row,
									robot.GetCol());
				}
			}
		}
	}
}

Robot *EnemyController::FindNearestHostile(const Robot &robot) const {
	Robot *nearest = nullptr;
	int nearest_distance = 0;
	for (Robot &candidate : robots_) {
		if (candidate.GetHealth() == 0 || robot.IsFriendlyTo(candidate)) {
			continue;
		}
		int distance = std::abs(candidate.GetRow() - robot.GetRow()) + std::abs(candidate.GetCol() - robot.GetCol());
		if (nearest == nullptr || distance < nearest_distance) {
			nearest = &candidate;
			nearest_distance = distance;
		}
	}
	return nearest;
}

Robot *EnemyController::FindRobotAt(int row, int col) const {
	for (Robot &robot : robots_) {
		if (robot.GetHealth() == 0) {
			continue;
		}
		if (robot.GetRow() == row && robot.GetCol() == col) {
			return &robot;
		}
	}
	return nullptr;
}

GameField::GameField(int width, int height) {
	if (width > kMinFieldSize && height > kMinFieldSize && width < kMaxFieldSize && height < kMaxFieldSize) {
		width_ = width;
		height_ = height;
		cells_.resize(height_);
		for (int row = 0; row < height_; ++row) {
			for (int col = 0; col < width_; ++col) {
				bool is_walkable = !(row == 0 || row == height_ - 1 || col == 0 || col == width_ - 1);
				cells_[row].push_back(Cell(is_walkable, kDefaultMovementCost));
			}
		}
	} else {
		throw std::invalid_argument("invalid size");
	}
}

const Cell &GameField::GetCell(int row, int col) const {
	if (!IsInside(row, col)) {
		throw std::invalid_argument("cell is outside the field");
	}
	return cells_[row][col];
}

void GameField::SetWalkable(int row, int col, bool is_walkable) {
	if (!IsInside(row, col)) {
		throw std::invalid_argument("cell is outside the field");
	}
	cells_[row][col].SetWalkable(is_walkable);
}


void GameField::SetWalkableArea(int top_row, int left_col, int height, int width, bool is_walkable) {
	if (height < 1 || width < 1) {
		throw std::invalid_argument("incorrect size");
	}
	if (!(IsInside(top_row, left_col) && IsInside(top_row + height - 1, left_col + width - 1))) {
		throw std::invalid_argument("area not inside");
	}
	for (int row = top_row; row <= top_row + height - 1; ++row) {
		for (int col = left_col; col <= left_col + width - 1; ++col) {
			SetWalkable(row, col, is_walkable);
		}
	}
}


void GameField::RevealCell(int row, int col) {
	if (!IsInside(row, col)) {
		throw std::invalid_argument("cell is outside the field");
	}
	cells_[row][col].SetKnown();
	cells_[row][col].SetVisible(true);
}


void GameField::SetMovementCost(int row, int col, int cost) {
	if (!IsInside(row, col)) {
		throw std::invalid_argument("cell is outside the field");
	}
	cells_[row][col].SetMovementCost(cost);
}

bool GameField::IsInside(int row, int col) const {
	return row >= 0 && row < height_ && col >= 0 && col < width_;
}

int GameField::GetWidth() const {
	return width_;
}

int GameField::GetHeight() const {
	return height_;
}

void GameField::ClearVisibility() {
	for (int row = 0; row < height_; ++row) {
		for (int col = 0; col < width_; ++col) {
			cells_[row][col].SetVisible(false);
		}
	}
}

GameLoop::GameLoop(GameField &game_field, std::vector<Robot> &robots, std::vector<RobotFactory> &factories)
	: game_field_(game_field),
	factories_(factories),
	robots_(robots),
	movement_system_(game_field_),
	enemy_controller_(robots_, movement_system_){
	for (const RobotFactory &factory: factories_) {
		factory.PlaceOn(game_field_);
	}
}


Robot &GameLoop::FindPlayerRobot() const {
	for (Robot &robot: robots_) {
		if (robot.GetTeam() == Team::kPlayer) {
			return robot;
		}
	}
	throw std::runtime_error("no player robot");
}




bool GameLoop::TryMoveRobot(Robot &robot, int target_row, int target_col) const {
	Robot *opponent = enemy_controller_.FindRobotAt(target_row, target_col);
	return movement_system_.Action(robot, opponent, target_row, target_col);
}

bool GameLoop::IsPlayerDefeated() const {
	return FindPlayerRobot().GetHealth() == 0;
}

void GameLoop::RegenerateEnergy() {
	for (Robot &robot: robots_) {
		robot.RegenerateEnergy(kRegenerateEnergy);
	}
}

void GameLoop::ResetSpeed() {
	for (Robot &robot: robots_) {
		robot.ResetSpeed();
	}
}


void GameLoop::Run() {
	while (!AllEnemiesDefeated() && !IsPlayerDefeated()) {
		PlayTurn();
	}
	Visualizer visualizer;
	visualizer.Draw(game_field_, robots_, factories_);
	if (IsPlayerDefeated()) {
		std::cout << "defeat" << std::endl;
	} else {
		std::cout << "victory" << std::endl;
	}
}


bool GameLoop::HandlePlayerInput() const {
	Robot &player = FindPlayerRobot();
	std::string input;
	std::cin >> input;
	if (input == kKeyEndTurn) {
		return true;
	}
	int row_offset = 0;
	int col_offset = 0;
	if (input == kKeyMoveUp) {
		row_offset = -1;
	} else if (input == kKeyMoveLeft) {
		col_offset = -1;
	} else if (input == kKeyMoveDown) {
		row_offset = 1;
	} else if (input == kKeyMoveRight) {
		col_offset = 1;
	} else {
		return false;
	}

	TryMoveRobot(player, player.GetRow() + row_offset,
				player.GetCol() + col_offset);
	return false;
}

bool GameLoop::AllEnemiesDefeated() const {
	for (const Robot &robot: robots_) {
		if (robot.GetTeam() == Team::kEnemy && robot.GetHealth() > 0) {
			return false;
		}
	}
	return true;
}

void GameLoop::PlayTurn() {
	ResetSpeed();
	RegenerateEnergy();
	RevealAroundPlayer();
	Visualizer visualizer;
	visualizer.Draw(game_field_, robots_, factories_);
	while (!HandlePlayerInput()) {
		RevealAroundPlayer();
		visualizer.Draw(game_field_, robots_, factories_);
	}
	enemy_controller_.AiTurn();
	SpawnEnemies();
}

void GameLoop::RevealAroundPlayer() {
	game_field_.ClearVisibility();
	Robot &player = FindPlayerRobot();
	int radius = player.GetVisionRadius();
	int row = player.GetRow();
	int col = player.GetCol();

	for (int dr = -radius; dr <= radius; ++dr) {
		for (int dc = -radius; dc <= radius; ++dc) {
			if ((dr * dr + dc * dc) <= radius * radius) {
				if (game_field_.IsInside(row + dr, col + dc)) {
					game_field_.RevealCell(row + dr, col + dc);
				}
			}
		}
	}
}

bool GameLoop::IsCellFree(int row, int col) const {
	return (game_field_.IsInside(row, col) && game_field_.GetCell(row, col).GetWalkable() && enemy_controller_.FindRobotAt(row, col) ==
			nullptr);
}




void GameLoop::SpawnEnemies() {
	for (RobotFactory &factory: factories_) {
		factory.EndTurn();
		if (!factory.IsReadyToSpawn()) {
			continue;
		}
		for (const std::pair<int, int> &cell: factory.GetAdjacentCells()) {
			if (IsCellFree(cell.first, cell.second)) {
				robots_.push_back(factory.CreateEnemy(cell.first, cell.second));
				break;
			}
		}
	}
}

MovementSystem::MovementSystem(GameField &game_field) : game_field_(game_field) {}

bool MovementSystem::Action(Robot &robot, Robot *opponent, int row, int col) const {
	if (!game_field_.IsInside(row, col)) {
		return false;
	}
	if (!game_field_.GetCell(row, col).GetWalkable()) {
		return false;
	}
	if (opponent != nullptr) {
		return robot.InteractWith(*opponent);
	}
	return robot.SetPosition(row, col, game_field_.GetCell(row, col).GetMovementCost());
}

Robot::Robot(int max_health, int attack_power, int max_energy, int experience_to_next_rank, Team team, int row, int col,
			int speed, int vision_radius) {
	if (max_health <= 0) {
		throw std::invalid_argument("incorrect max_health value");
	}
	if (attack_power < 0) {
		throw std::invalid_argument("incorrect attack_power value");
	}
	if (max_energy < kActionCost) {
		throw std::invalid_argument("incorrect max_energy value");
	}
	if (experience_to_next_rank <= 0) {
		throw std::invalid_argument("incorrect experience_to_next_rank value");
	}
	if (speed < kMinSpeed) {
		throw std::invalid_argument("incorrect speed value");
	}
	if (row < 0 || col < 0) {
		throw std::invalid_argument("incorrect position value");
	}
	if (vision_radius <= 0) {
		throw std::invalid_argument("incorrect vision_radius value");
	}
	max_health_ = max_health;
	health_ = max_health;
	energy_ = max_energy;
	max_energy_ = max_energy;
	experience_ = 0;
	attack_power_ = attack_power;
	rank_ = kInitialRank;
	experience_to_next_rank_ = experience_to_next_rank;
	team_ = team;
	row_ = row;
	col_ = col;
	speed_ = speed;
	remaining_speed_ = speed;
	vision_radius_ = vision_radius;
}


int Robot::GetExperience() const {
	return experience_;
}

int Robot::GetExperienceToNextRank() const {
	return experience_to_next_rank_;

int Robot::Clamp(int value, int min_value, int max_value) {
	if (value < min_value) {
		return min_value;
	}
	if (value > max_value) {
		return max_value;
	}
	return value;
}

void Robot::TakeDamage(int amount) {
	if (amount < 0) {
		throw std::invalid_argument("damage must not be negative");
	}
	health_ = Clamp(health_ - amount, 0, max_health_);
}

void Robot::Heal(int amount) {
	if (amount < 0) {
		throw std::invalid_argument("heal amount must not be negative");
	}
	health_ = Clamp(health_ + amount, 0, max_health_);
}

bool Robot::InteractWith(Robot &other) {
	if (SpendEnergy(kActionCost)) {
		if (IsFriendlyTo(other)) {
			other.Heal(attack_power_);
		} else {
			other.TakeDamage(attack_power_);
			if (other.GetHealth() == 0) {
				GainExperience(kExperienceForKill);
			}
		}
		return true;
	}
	return false;
}


void Robot::RankUp() {
	rank_ += 1;
	max_health_ += kRankUpHealthBonus;
	attack_power_ += kRankUpAttackBonus;
	max_energy_ += kRankUpEnergyBonus;
	experience_to_next_rank_ *= kRankUpExperienceMultiplier;
	health_ = max_health_;
	energy_ = max_energy_;
}

bool Robot::SetPosition(int row, int col, int cost) {
	if (row < 0 || col < 0) {
		throw std::invalid_argument("position must not be negative");
	}
	if (cost < kMinMovementCost) {
		throw std::invalid_argument("movement cost is below the minimum");
	}
	if (energy_ >= kActionCost && remaining_speed_ >= cost) {
		SpendEnergy(kActionCost);
		SpendSpeed(cost);
		row_ = row;
		col_ = col;
		return true;
	}

	return false;
}

bool Robot::SpendEnergy(int amount) {
	if (amount > energy_) {
		return false;
	}
	energy_ -= amount;
	return true;
}

int Robot::GetRemainingSpeed() const {
	return remaining_speed_;
}

int Robot::GetSpeed() const {
	return speed_;
}

bool Robot::SpendSpeed(int amount) {
	if (remaining_speed_ >= amount) {
		remaining_speed_ -= amount;
		return true;
	}
	return false;
}


void Robot::ResetSpeed() {
	remaining_speed_ = speed_;
}

void Robot::GainExperience(int amount) {
	if (amount < 0) {
		throw std::invalid_argument("experience must not be negative");
	}
	experience_ += amount;
	while (experience_ >= experience_to_next_rank_) {
		RankUp();
	}
}

int Robot::GetVisionRadius() const {
	return vision_radius_;
}

void Robot::SetVisionRadius(int vision_radius) {
	if (vision_radius <= 0) {
		throw std::invalid_argument("vision_radius must be positive");
	}
	vision_radius_ = vision_radius;
}


void Robot::RegenerateEnergy(int amount) {
	if (amount < 0) {
		throw std::invalid_argument("energy amount must not be negative");
	}
	energy_ = Clamp(energy_ + amount, 0, max_energy_);
}


bool Robot::IsFriendlyTo(const Robot &other) const {
	Team other_team = other.GetTeam();
	return team_ == other_team ||
			(team_ != Team::kEnemy && other_team != Team::kEnemy);
}

void Robot::SetMaxHealth(int max_health) {
	if (max_health <= 0) {
		throw std::invalid_argument("max_health must be positive");
	}
	max_health_ = max_health;
	if (health_ > max_health_) {
		health_ = max_health_;
	}
}

void Robot::SetAttackPower(int attack_power) {
	if (attack_power < 0) {
		throw std::invalid_argument("attack_power must not be negative");
	}
	attack_power_ = attack_power;
}

void Robot::SetMaxEnergy(int max_energy) {
	if (max_energy < kActionCost) {
		throw std::invalid_argument("max_energy must be at least kActionCost");
	}
	max_energy_ = max_energy;
	if (energy_ > max_energy_) {
		energy_ = max_energy_;
	}

void Robot::SetExperienceToNextRank(int experience_to_next_rank) {
	if (experience_to_next_rank <= 0) {
		throw std::invalid_argument("experience_to_next_rank must be positive");
	}
	experience_to_next_rank_ = experience_to_next_rank;
}


int Robot::GetRank() const {
	return rank_;
}

void Robot::SetSpeed(int speed) {
	if (speed < kMinSpeed) {
		throw std::invalid_argument("speed must be at least kMinSpeed");
	}
	speed_ = speed;
	if (remaining_speed_ > speed_) {
		remaining_speed_ = speed_;
	}
}

RobotFactory::RobotFactory(int row, int col, int turns_between_spawns) {
	if (row < 0 || col < 0) {
		throw std::invalid_argument("factory position must not be negative");
	}
	if (turns_between_spawns <= 0) {
		throw std::invalid_argument("turns_between_spawns must be positive");
	}
	row_ = row;
	col_ = col;
	turns_between_spawns_ = turns_between_spawns;
}


int RobotFactory::GetRow() const {
	return row_;
}


void RobotFactory::EndTurn() {
	if (turns_since_spawn_ < turns_between_spawns_) {
		turns_since_spawn_++;
	}
}


Robot RobotFactory::CreateEnemy(int row, int col) {
	if (IsReadyToSpawn()) {
		turns_since_spawn_ = 0;
		return Robot(kSpawnedMaxHealth, kSpawnedAttackPower, kSpawnedMaxEnergy, kSpawnedExperienceToNextRank,
					Team::kEnemy, row, col, kSpawnedSpeed, kVisionRadius);
	}
	throw std::logic_error("cant create robot");
}

bool RobotFactory::Occupies(int row, int col) const {
	return row >= row_ && row < row_ + kFactorySize && col >= col_ &&
			col < col_ + kFactorySize;
}


std::vector<std::pair<int, int> > RobotFactory::GetAdjacentCells() const {
	std::vector<std::pair<int, int> > cells;
	for (int row = row_ - 1; row < row_ + kFactorySize + 1; row++) {
		for (int col = col_ - 1; col < col_ + kFactorySize + 1; col++) {
			if (Occupies(row, col)) {
				continue;
			}
			cells.push_back({row, col});
		}
	}
	return cells;
}

void RobotFactory::PlaceOn(GameField &field) const {
	field.SetWalkableArea(row_, col_, kFactorySize, kFactorySize, false);
}


bool RobotFactory::IsReadyToSpawn() const {
	return turns_since_spawn_ >= turns_between_spawns_;
}

int RobotFactory::GetCol() const {
	return col_;
}

void Visualizer::Draw(const GameField &field, const std::vector<Robot> &robots,
					const std::vector<RobotFactory> &factories) const {
	std::cout << kClearScreenSequence;
	for (int row = 0; row < field.GetHeight(); ++row) {
		for (int col = 0; col < field.GetWidth(); ++col) {
			if (!field.GetCell(row, col).GetKnown()) {
				std::cout << kUnknownCellSymbol;
				continue;
			}
			if (IsFactoryAt(factories, row, col)) {
				std::cout << kFactorySymbol;
				continue;
			}
			bool is_visible = field.GetCell(row, col).GetVisible();
			const Robot *robot_here = nullptr;
			for (const Robot &robot: robots) {
				if (robot.GetRow() == row && robot.GetCol() == col) {
					if (robot.GetHealth() == 0 || !is_visible) {
						continue;
					}
					robot_here = &robot;
					break;
				}
			}
			if (robot_here != nullptr) {
				std::cout << GetRobotSymbol(robot_here->GetTeam());
			} else {
				const Cell &cell = field.GetCell(row, col);
				if (!cell.GetWalkable()) {
					std::cout << kWallSymbol;
				} else if (cell.GetMovementCost() == kSlowCellThreshold) {
					std::cout << kSlowCellSymbol;
				} else {
					std::cout << kFloorSymbol;
				}
			}
		}
		std::cout << '\n';
	}
	for (const Robot &robot: robots) {
		if (robot.GetHealth() == 0 || !field.GetCell(robot.GetRow(), robot.GetCol()).GetVisible()) {
			continue;
		}
		std::cout << GetRobotSymbol(robot.GetTeam())
				<< " HP: " << robot.GetHealth() << "/" << robot.GetMaxHealth()
				<< " Energy: " << robot.GetEnergy() << "/" << robot.GetMaxEnergy()
				<< " Speed: " << robot.GetRemainingSpeed() << "/" << robot.GetSpeed()
				<< " Rank: " << robot.GetRank()
				<< " EXP: " << robot.GetExperience() << "/" << robot.GetExperienceToNextRank()
				<< std::endl;
	}
}


bool Visualizer::IsFactoryAt(const std::vector<RobotFactory> &factories, int row, int col) const {
	for (const RobotFactory &factory: factories) {
		if (factory.Occupies(row, col)) {
			return true;
		}
	}
	return false;
}

const char *Visualizer::GetRobotSymbol(Team team) const {
	switch (team) {
		case Team::kPlayer:
			return kPlayerSymbol;
		case Team::kAlly:
			return kAllySymbol;
		case Team::kEnemy:
			return kEnemySymbol;
	}
	return kEnemySymbol;
}
